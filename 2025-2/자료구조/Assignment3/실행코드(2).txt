import matplotlib.pyplot as plt

# Helper for plotting
def plot_metric(df, graph_type: str, metric: str, title_suffix: str, ylabel: str, filename: str):
    sub = df[df["Type"] == graph_type]
    Ns_sorted = sorted(sub["N"].unique())

    mat_vals = []
    list_vals = []
    for n in Ns_sorted:
        mat_vals.append(float(sub[(sub["N"] == n) & (sub["Rep"] == "Matrix")][metric]))
        list_vals.append(float(sub[(sub["N"] == n) & (sub["Rep"] == "List")][metric]))

    plt.figure()
    plt.plot(Ns_sorted, mat_vals, marker="o", label="Adjacency Matrix")
    plt.plot(Ns_sorted, list_vals, marker="o", label="Adjacency List")
    plt.title(f"{graph_type.capitalize()} graph - {title_suffix}")
    plt.xlabel("N (number of nodes)")
    plt.ylabel(ylabel)
    plt.legend()
    plt.grid(True)
    plt.tight_layout()
    plt.show()
